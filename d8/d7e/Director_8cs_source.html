72 /// Initialise the director and connect to OpenAI's server.
73 /// <param name="onDirectorResponse">The callback to invoke when the director has a response ready.</param>
74 /// <param name="cancellationToken">Cancellation token used to cancel any async actions when the program shuts down.</param>
75 public void Initialise(UnityAction<Director.Response> onDirectorResponse, CancellationToken cancellationToken) {
76 // Set the callback to inform the VoiceChat system whenever a director response is ready.
77 this.onDirectorResponse = onDirectorResponse;
78
79 // Run the director session in a separate thread.
80 Func<CancellationToken, Task> run = async (CancellationToken cancellationToken) => {
81 try {
82 // Create session.
83 session = await api.RealtimeEndpoint.CreateSessionAsync(GetSessionConfiguration(), cancellationToken);
84 // Start recording user audio.
85 RecordInputAudio(session, cancellationToken);
86 // Let the VoiceChat system know we are ready to receive input.
87 IsConnected = true;
88 // Receive server response in a loop.
89 await session.ReceiveUpdatesAsync<IServerEvent>(OnServerEvent, cancellationToken);
90 } catch (Exception e) {
91 switch (e) {
92 case TaskCanceledException: break;
93 case OperationCanceledException: break;
94 default: Debug.LogException(e); break;
95 }
96 } finally {
97 session?.Dispose();
98 session = null;
99 Debug.Log("Director's session disposed.");
100 }
101 };
102 _ = run(cancellationToken);
103 }

107 /// Listens for the next user input. This needs to be invoked at the start of every round, in order to let the director prepare for user input.
108 /// Before invoking, you need to check that the status is Waiting.
109 /// <param name="config">The director configuration.</param>
110 /// <param name="speakers">The list of actors that are possibly speaking. The speaker order will be determined by choosing actors from this list.</param>
111 /// <param name="topics">Possible discussion topics to be triggered by the user input.</param>
112 /// <param name="cancellationToken">Cancellation token used to cancel any async actions when the program shuts down.</param>
113 public async void ListenForNextUserInput(DirectorConfig config, List<string> speakers, List<string> topics, CancellationToken cancellationToken) {
114 if (!IsStatus(Status.Waiting)) {
115 throw new System.Exception("Director.ListenForNextInput can only be invoked if the status is Waiting!");
116 }
117
118 // Update director session configuration.
119 this.config = config;
120 this.tools = new List<Tool>() { BuildTriggerResponseTool(speakers), BuildTriggerDiscussionTool(topics) };
121 this.response = new Director.Response();
122
123 try {
124 await session.SendAsync(new UpdateSessionRequest(GetSessionConfiguration()), cancellationToken);
125 } catch (Exception e) {
126 Debug.LogException(e);
127 }
128
129 // Starting listening to the human user.
130 SetStatus(Status.Listening);
131 }

143 /// Submit the user text input. Used as an alternative to speaking into the microphone, usually for development & debugging purposes.
144 /// <param name="message">The user text input.</param>
145 /// <param name="cancellationToken">Cancellation token used to cancel any async actions when the program shuts down.</param>
146 public async Awaitable<bool> SubmitUserTextInput(string message, CancellationToken cancellationToken) {
147 // Check for status and update it.
148 if (!TestAndSetStatus(Status.Listening, Status.TextInput)) { return false; }
149
150 // Tell the director to clear everything it has heard.
151 await session.SendAsync(new InputAudioBufferClearRequest(), cancellationToken);
152
153 // Now tell it to reply to our text input.
154 response.userTranscript = message;
155 await session.SendAsync(new OpenAI.Realtime.ConversationItemCreateRequest(message), cancellationToken);
156 await session.SendAsync(new OpenAI.Realtime.CreateResponseRequest(), cancellationToken);
157
158 return true;
159 }

173 private async void RecordInputAudio(OpenAI.Realtime.RealtimeSession session, CancellationToken cancellationToken) {
174 var memoryStream = new MemoryStream();
175 var semaphore = new SemaphoreSlim(1, 1);
176
177 try {
178 byte[] emptyBuffer = new byte[1024 * 16]; // 1 KB buffer.
179 async Task BufferCallback(NativeArray<byte> buffer) {
180 try {
181 await semaphore.WaitAsync(CancellationToken.None).ConfigureAwait(false);
182 for (int i = 0; i < buffer.Length; ++i) { memoryStream.WriteByte(buffer[i]); }
183 } finally {
184 semaphore.Release();
185 }
186 }
187
188 // RecordingManager is from the com.utilities.audio package.
189 // We don't await this so that we can implement buffer copy and send response to realtime API.
190 RecordingManager.StartRecordingStream<WavEncoder>(BufferCallback, 24000, cancellationToken); // Sample rate has to be 24000 according to the InputAudioBufferAppendRequest API docs.
191
192 do {
193 byte[] voiceBuffer = ArrayPool<byte>.Shared.Rent(1024 * 16); // 16 KB buffer.
194 try {
195 int bytesRead = 0;
196 try {
197 await semaphore.WaitAsync(cancellationToken).ConfigureAwait(false);
198 memoryStream.Position = 0;
199 bytesRead = await memoryStream.ReadAsync(voiceBuffer, 0, (int)Math.Min(voiceBuffer.Length, memoryStream.Length), cancellationToken).ConfigureAwait(false);
200 memoryStream.SetLength(0);
201 } finally {
202 semaphore.Release();
203 }
204
205 if (bytesRead > 0) {
206 // If we are recording, send what the microphone picks up.
207 if (!IsMicMuted && IsStatus(Status.Listening)) {
208 await session.SendAsync(new InputAudioBufferAppendRequest(voiceBuffer.AsMemory(0, bytesRead)), cancellationToken).ConfigureAwait(false);
209 }
210 // Otherwise, send silence. We want to continue sending data so that the model can trigger a response if it received silence.
211 else {
212 await session.SendAsync(new InputAudioBufferAppendRequest(emptyBuffer.AsMemory(0, bytesRead)), cancellationToken).ConfigureAwait(false);
213 }
214 } else {
215 await Task.Yield();
216 }
217 } catch (Exception e) {
218 switch (e) {
219 // Ignored
220 case TaskCanceledException: break;
221 case OperationCanceledException: break;
222 default: Debug.LogError(e); break;
223 }
224 } finally {
225 ArrayPool<byte>.Shared.Return(voiceBuffer);
226 }
227 } while (!cancellationToken.IsCancellationRequested);
228
229 RecordingManager.EndRecording();
230 } catch (Exception e) {
231 switch (e) {
232 // Ignored
233 case TaskCanceledException: break;
234 case OperationCanceledException: break;
235 default: Debug.LogError(e); break;
236 }
237 } finally {
238 await memoryStream.DisposeAsync();
239 }
240 }

255 /// Callback function to receive server events from OpenAI.
256 /// <param name="event">The event received from OpenAI.</param>
257 private void OnServerEvent(IServerEvent @event) {
258 switch (@event) {
259 case RealtimeEventError error: throw error;
260 case SessionResponse sessionResponse: break;
261 case RealtimeConversationResponse conversationResponse: break;
262 case ConversationItemCreatedResponse conversationItemCreated: break;
263 case ConversationItemInputAudioTranscriptionResponse conversationItemTranscription:
264 if (!conversationItemTranscription.IsCompleted) { return; }
265 Debug.Log($"[{conversationItemTranscription.ItemId}] Director's User Transcription: " + conversationItemTranscription.Transcript.Trim());
266
267 // We only want to update the user transcript if this is the latest transcript.
268 // This is to handle the insane case where:
269 // 1. User sends text and voice input at the same time.
270 // 2. Voice input is ignored, but a InputAudioBufferCommittedResponse was received.
271 // 3. ConversationItemInputAudioTranscriptionResponse was not received.
272 // 4. User sends voice input.
273 // 5. Second InputAudioBufferCommittedResponse is received.
274 // 6. First outdated ConversationItemInputAudioTranscriptionResponse is received. (This needs to be ignored.)
275 // 7. Second correct ConversationItemInputAudioTranscriptionResponse is received.
276 lock (latestItemId) {
277 if (latestItemId != conversationItemTranscription.ItemId) {
278 Debug.Log("Director.OnServerEvent ConversationItemInputAudioTranscriptionResponse ignored as it is outdated.");
279 }
280 }
281
282 if (IsStatus(Status.VoiceInput)) {
283 response.userTranscript = conversationItemTranscription.Transcript.Trim();
284 InvokeOnDirectorResponse();
285 } else {
286 Debug.Log("Director.OnServerEvent ConversationItemInputAudioTranscriptionResponse ignored as the status is not ReplyToVoice.");
287 }
288 break;
289 case ConversationItemTruncatedResponse conversationItemTruncated: break;
290 case ConversationItemDeletedResponse conversationItemDeleted: break;
291 case InputAudioBufferCommittedResponse committedResponse:
292 Debug.Log($"[{committedResponse.ItemId}] Director.OnServerEvent InputAudioBufferCommittedResponse");
293
294 // InputAudioBufferCommittedResponse should always be received before ConversationItemInputAudioTranscriptionResponse.
295 lock (latestItemId) { latestItemId = committedResponse.ItemId; }
296
297 if (TestAndSetStatus(Status.Listening, Status.VoiceInput)) {
298 session.Send(new OpenAI.Realtime.CreateResponseRequest()); // Now tell it to reply to our audio input.
299 } else {
300 Debug.Log("Director.OnServerEvent InputAudioBufferCommittedResponse ignored.");
301 }
302 break;
303 case InputAudioBufferClearedResponse clearedResponse:
304 Debug.Log($"Director.OnServerEvent InputAudioBufferClearedResponse");
305 break;
306 case InputAudioBufferStartedResponse startedResponse:
307 Debug.Log($"Director.OnServerEvent InputAudioBufferStartedResponse");
308 break;
309 case InputAudioBufferStoppedResponse stoppedResponse: break;
310 case RealtimeResponse realtimeResponse:
311 switch (realtimeResponse.Response.Status) {
312 case RealtimeResponseStatus.InProgress:
313 Debug.Log("Director Realtime Response InProgress.");
314 break;
315 case RealtimeResponseStatus.Completed:
316 Debug.Log("Director Realtime Response Completed.");
317 InvokeOnDirectorResponse();
318 break;
319 case RealtimeResponseStatus.Cancelled:
320 Debug.Log("Director Realtime Response Cancelled.");
321 break;
322 case RealtimeResponseStatus.Failed:
323 Debug.Log("Director Realtime Response Failed.");
324 break;
325 case RealtimeResponseStatus.Incomplete:
326 Debug.Log("Director Realtime Response Incomplete.");
327 break;
328 }
329 break;
330 case ResponseOutputItemResponse outputItemResponse: break;
331 case ResponseContentPartResponse contentPartResponse: break;
332 case ResponseTextResponse textResponse: break; // Used if modality is Modality.Text only.
333 case ResponseAudioResponse audioResponse: break;
334 case ResponseAudioTranscriptResponse transcriptResponse: break; // Used if modality has Modality.Audio.
335 case ResponseFunctionCallArgumentsResponse functionCallArgumentsResponse:
336 if (!functionCallArgumentsResponse.IsDone) { return; }
337 Debug.Log($"Director's Function Call: " + functionCallArgumentsResponse.Name + ", Arguments: " + functionCallArgumentsResponse.Arguments.ToString());
338
339 // Handle function calls.
340 string output = string.Empty;
341 if (functionCallArgumentsResponse.Name == "trigger_discussion") {
342 response.discussionTopic = ParseDiscussionTopic(functionCallArgumentsResponse.Arguments.ToString());
343 } else if (functionCallArgumentsResponse.Name == "trigger_response") {
344 response.speakerOrder = ParseSpeakerOrder(functionCallArgumentsResponse.Arguments.ToString());
345 }
346
347 // Return the function call output to the model.
348 ConversationItem functionCallOutput = new ConversationItem((ToolCall)functionCallArgumentsResponse, output);
349 session.Send(new OpenAI.Realtime.ConversationItemCreateRequest(functionCallOutput));
350 break;
351 case RateLimitsResponse rateLimitsResponse: break;
352 default: break;
353 }
354 }

369 /// Create a function following OpenAI's JSON Schema for the director to decide upon the speaker order.
370 /// OpenAI API on function calling: https://platform.openai.com/docs/guides/function-calling
371 /// <param name="speakers">The names of the speakers.</param>
372 /// <returns>The function's JSON Object.</returns>
373 private OpenAI.Function BuildTriggerResponseTool(List<string> speakers) {
374 var args = new {
375 type = "object",
376 properties = new {
377 speaker_order = new {
378 type = "array",
379 description = "The order which the AI models will respond to the user.",
380 items = new {
381 type = "string",
382 @enum = speakers.ToArray() // Adding an enum means that the AI can only pick from this set of values. (Well, the AI is stupid and still sometimes hallucinates invalid values.)
383 },
384 minItems = 1,
385 maxItems = speakers.Count,
386 uniqueItems = true
387 }
388 },
389 required = new[] { "speaker_order" }
390 };
391 string parameters = JsonConvert.SerializeObject(args, Formatting.Indented);
392 return new OpenAI.Function("trigger_response", "Triggers the AI models to respond to the user. No output is given.", JToken.Parse(parameters));
393 }
