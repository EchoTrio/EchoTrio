109 /// Append an assistant message to the actor's conversation history. This means that we can make the actor think it said something, even if it did not.
110 /// Used for scripted speech, where we force the actor to say something the designer wrote.
111 /// <param name="message">The message to append.</param>
112 /// <param name="emotion">The emotion of the message.</param>
113 /// <param name="cancellationToken">Cancellation token used to cancel any async actions when the program shuts down.</param>
114 /// <returns>The actor's response.</returns>
115 public async Task<Actor.Response> InsertResponse(string message, Emotion emotion, CancellationToken cancellationToken) {
116 // Bit of a hack because assistant messages must now be of type output_text. Had to make my own custom class because the package's creator rejected my pull request.
117 conversation.Add(new Message(OpenAI.Role.Assistant, new OutputTextContent(message)));
118 return new Actor.Response() {
119 message = message,
120 emotion = emotion,
121 audioClip = await GetAudioClipAsync(message, cancellationToken)
122 };
123 }

125 /// Request the actor to generate a response based on the conversation history.
126 /// <param name="cancellationToken">Cancellation token used to cancel any async actions when the program shuts down.</param>
127 /// <returns>The actor's response.</returns>
128 public async Task<Actor.Response> GetResponse(CancellationToken cancellationToken) {
129 try {
130 // Request a response from OpenAI.
131 CreateResponseRequest request = new CreateResponseRequest(
132 input: conversation,
133 model: openAISettings.model,
134 tools: openAISettings.tools,
135 reasoning: new Reasoning(openAISettings.reasoningEffort, OpenAI.ReasoningSummary.Auto),
136 maxToolCalls: 0 < openAISettings.tools.Count ? openAISettings.tools.Count : null,
137 include: openAISettings.include);
138 OpenAI.Responses.Response response = await openAIApi.ResponsesEndpoint.CreateModelResponseAsync(request, cancellationToken: cancellationToken);
139
140 // Get response from OpenAI.
141 Actor.Response actorResponse = new Actor.Response();
142 for (int i = 0; i < response.Output.Count; ++i) {
143 IResponseItem responseItem = response.Output[i];
144 switch (responseItem) {
145 case OpenAI.Responses.Message message:
146 conversation.Add(message);
147 actorResponse.message = message.ToString();
148 actorResponse.audioClip = await GetAudioClipAsync(message.ToString(), cancellationToken);
149 break;
150 case OpenAI.Responses.ReasoningItem reasoningItem:
151 conversation.Add(reasoningItem);
152 List<string> reasonings = new List<string>();
153 foreach (OpenAI.Responses.ReasoningSummary reasoningSummary in reasoningItem.Summary) {
154 actorResponse.reasonings.Add(reasoningSummary.Text);
155 }
156 break;
157 case OpenAI.Responses.WebSearchToolCall webSearchToolCall:
158 Debug.Log("Actor " + Persona + " Searched Web");
159 break;
160 case OpenAI.Responses.FileSearchToolCall fileSearchToolCall:
161 Debug.Log("Actor " + Persona + " Searched Files");
162 break;
163 case OpenAI.Responses.FunctionToolCall functionToolCall:
164 Debug.Log("Actor " + Persona + " Function Call: " + functionToolCall.Name + ", Arguments: " + functionToolCall.Arguments.ToString());
165
166 // Handle function calls.
167 string output = string.Empty;
168 if (functionToolCall.Name == "set_emotion") {
169 output = ParseEmotion(functionToolCall.Arguments.ToString());
170 actorResponse.emotion = output.ToEmotion();
171 }
172
173 // Return the function call output to the model.
174 conversation.Add(functionToolCall);
175 conversation.Add(new FunctionToolCallOutput(functionToolCall, output));
176 return await GetResponse(cancellationToken);
177 default:
178 Debug.LogWarning("Actor.GetResponse: Unhandled " + responseItem.GetType().Name + " Received");
179 break;
180 }
181 }
182 return actorResponse;
183 } catch (Exception e) {
184 Debug.Log(e);
185 return null;
186 }
187 }

190 private async Task<AudioClip> GetAudioClipAsync(string text, CancellationToken cancellationToken) {
191 try {
192 ElevenLabs.Voices.Voice voice = new ElevenLabs.Voices.Voice(elevenLabsSettings.voiceId, this.Persona);
193 TextToSpeechRequest request = new TextToSpeechRequest(
194 voice, text,
195 model: ContainsAudioTags(text) ? elevenLabsSettings.expressionModel : elevenLabsSettings.fastModel,
196 languageCode: elevenLabsSettings.languageCode,
197 outputFormat: OutputFormat.PCM_24000); // Output format must be PCM, because that's what the AudioClip convertor in VoiceClip expects.
198 ElevenLabs.VoiceClip voiceClip = await elevenLabsApi.TextToSpeechEndpoint.TextToSpeechAsync(request, cancellationToken);
199 return voiceClip.AudioClip;
200 } catch (Exception e) {
201 Debug.Log(e);
202 }
203 return null;
204 }

211 /// Create a function following OpenAI's JSON Schema for the actor to select its emotion for the current response.
212 /// OpenAI API on function calling: https://platform.openai.com/docs/guides/function-calling
213 /// <returns>The function's JSON Object.</returns>
214 private OpenAI.Function BuildSetEmotionTool() {
215 List<string> emotions = new List<string>();
216 for (int i = 0; i < (int)Emotion.Num; ++i) {
217 Emotion emotion = (Emotion)i;
218 emotions.Add(emotion.ToString());
219 }
220
221 var args = new {
222 type = "object",
223 properties = new {
224 emotion = new {
225 type = "string",
226 description = "The emotion of your reply.",
227 @enum = emotions.ToArray() // Adding an enum means that the AI can only pick from this set of values. (Well, the AI is stupid and still sometimes hallucinates invalid values.)
228 }
229 },
230 required = new[] { "emotion" }
231 };
232 string parameters = JsonConvert.SerializeObject(args, Formatting.Indented);
233 return new OpenAI.Function("set_emotion", "Set the emotion of your current reply. The selected emotion is returned.", JToken.Parse(parameters));
234 }
235
236 private string ParseEmotion(string args) {
237 JToken parsedArgs = JToken.Parse(args);
238 if (parsedArgs == null || parsedArgs["emotion"] == null) { return string.Empty; }
239 string emotion = parsedArgs["emotion"].ToString();
240 return emotion;
241 }
