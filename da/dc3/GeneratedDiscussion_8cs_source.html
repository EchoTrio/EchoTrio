16 public List<Persona> GenerateRandomSpeakerOrder() {
17 if (speakers.Count == 0) { return new List<Persona>(); }
18
19 int numTurns = UnityEngine.Random.Range(minTurns, maxTurns + 1);
20 if (speakers.Count == 1) { return Enumerable.Repeat(speakers[0], numTurns).ToList(); }
21
22 List<Persona> order = new List<Persona>();
23 int prevIndex = -1;
24 int currIndex = -1;
25 for (int i = 0; i < numTurns; ++i) {
26 currIndex = UnityEngine.Random.Range(0, speakers.Count);
27 // If the current speaker is same as the previous, get the next speaker in line instead.
28 currIndex = (currIndex == prevIndex) ? (currIndex + 1) % speakers.Count : currIndex;
29 prevIndex = currIndex;
30 order.Add(speakers[currIndex]);
31 }
32 return order;
33 }
