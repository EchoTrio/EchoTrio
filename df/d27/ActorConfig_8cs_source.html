64 /// <summary>
65 /// Overrides any actor config if value is set in ActorOverrides.ini.
66 /// </summary>
67 /// <returns>Returns a copy of this ActorConfig with values overridden.</returns>
68 public ActorConfig Override() {
69 ActorConfig overriddenConfig = Instantiate(this);
70
71 string filePath = $"{Application.streamingAssetsPath}/Configs/{OverrideFileName}";
72 IConfiguration config = new ConfigurationBuilder().AddIniFile(filePath).Build();
73 IConfigurationSection section = config.GetSection(persona.ToString());
74
75 // We cannot proceed if the section does not exist.
76 if (section == null) {
77 Debug.LogWarning($"Section {persona.ToString()} not found in {filePath}!");
78 return overriddenConfig;
79 }
80
81 // Define a helper function.
82 Func<string, string> GetValue = (string key) => { return section[key] == null ? string.Empty : section[key].Trim(); };
83 string value = string.Empty;
84
85 // Override OpenAI Vector Store ID
86 value = GetValue("openai_vector_store_id");
87 if (!string.IsNullOrEmpty(value)) {
88 Debug.Log($"Overrode {persona.ToString()}'s OpenAI Vector Store ID.");
89 overriddenConfig.openAIVectorStoreId = value;
90 }
91
92 // Override ElevenLabs Voice ID
93 value = GetValue("elevenlabs_voice_id");
94 if (!string.IsNullOrEmpty(value)) {
95 Debug.Log($"Overrode {persona.ToString()}'s ElevenLabs Voice ID.");
96 overriddenConfig.elevenLabsVoiceId = value.Trim();
97 }
98
99 // Override Features
100 value = GetValue("feature_web_search");
101 if (!string.IsNullOrEmpty(value) && value.ToUpper() == "TRUE") {
102 Debug.Log($"Overrode {persona.ToString()}'s Web Search Feature Setting.");
103 overriddenConfig.enabledFeatures |= Feature.WebSearch;
104 }
105
106 value = GetValue("feature_file_search");
107 if (!string.IsNullOrEmpty(value) && value.ToUpper() == "TRUE") {
108 Debug.Log($"Overrode {persona.ToString()}'s File Search Feature Setting.");
109 overriddenConfig.enabledFeatures |= Feature.FileSearch;
110 }
111
112 value = GetValue("feature_reasoning");
113 if (!string.IsNullOrEmpty(value) && value.ToUpper() == "TRUE") {
114 Debug.Log($"Overrode {persona.ToString()}'s Reasoning Feature Setting.");
115 overriddenConfig.enabledFeatures |= Feature.Reasoning;
116 }
117
118 value = GetValue("feature_audio_tags");
119 if (!string.IsNullOrEmpty(value) && value.ToUpper() == "TRUE") {
120 Debug.Log($"Overrode {persona.ToString()}'s Audio Tags Feature Setting.");
121 overriddenConfig.enabledFeatures |= Feature.AudioTags;
122 }
123
124 return overriddenConfig;
125 }

127 // Internal Variables
128 private string GetSnippetsInstructions(string prefix, InstructionSnippet[] snippets) {
129 // Check if there are any valid snippets.
130 bool hasValidSnippets = false;
131 foreach (InstructionSnippet snippet in snippets) {
132 if (snippet == null) { continue; }
133 hasValidSnippets = hasValidSnippets || snippet.enabled;
134 }
135 if (!hasValidSnippets) { return string.Empty; }
136
137 // Combine all the snippets into a single string.
138 string instructions = $"{prefix}\n";
139 foreach (InstructionSnippet snippet in snippets) {
140 if (snippet == null) { continue; }
141 if (!snippet.enabled) { continue; }
142 instructions += $"{snippet.instruction}\n";
143 }
144 return instructions + "\n";
145 }
