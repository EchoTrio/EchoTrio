11 /// Voice chat system that acts as an intermediary between the human user and the AI models.
12 /// The system works using the concept of rounds. During every round, a few things may happen.
13 /// - A scripted discussion is triggered at the start of the round specified by the designer where the actors speak scripted lines. No user input is allowed in this round. OR
14 /// - A generated discussion is triggered at the start of the round specified by the designer where the actors speak AI generated lines. No user input is allowed in this round. OR
15 /// - User input is accepted and the actors generates a reply. OR
16 /// - User input is accepted and it triggers a scripted or generated discussion as the reply from the actors if the user mentions certain topics. OR
17 /// - User input is allowed by the user does not provide any input. A scripted or generated dicussion is triggered after some time as specified by the designer.
18 public class VoiceChat : MonoBehaviour {

21 /// All the possible states of the system.
22 private enum State {
23 Invalid = -1,
24 /// Idle while waiting for player to start the game.
25 Idle,
26 /// Before we start each round, we enter the Prepare stage, and make a decision of which state to enter for the round.
27 Prepare,
28 /// If user input is allowed this round, enter the Wait stage to wait for the actors to finish speaking, and ensure that the director has connected to OpenAI's server.
29 Wait,
30 /// Listen for user input.
31 Listen,
32 /// Actors responding to user input as per normal AI conversations.
33 FreeTalk,
34 /// Actors playing a scripted discussion, or generating a discussion on a topic.
35 Discuss,
36 /// The chat has ended.
37 Finish,
38 Num
39 }

240 /// <summary>
241 /// Queue up an actor's output to be played by the audio thread.
242 /// </summary>
243 /// <param name="actorOutput">The actor output to queue.</param>
244 private async Awaitable QueueActorOutput(string persona, string message, Emotion emotion, List<string> reasonings, AudioStream audioStream, AudioSource audioSource) {
245 // Flag that audio has started playing.
246 // Even though it doesn't actually play until the audio thread plays it, this flag has to be set here and not in the audio thread to ensure that the main thread doesn't start listening to user input again.
247 // Because in theory it can be a few frames before the audio thread gets to it and we don't want to accidentally trigger listening for user input in the mean time.
248 isAudioPlaying = true;
249
250 // Ensure that the audio clip is added to the garbage collector before being queued.
251 // This is so that it is impossible for the audio thread to mark a clip as finished and putting it into the garbage collector before we do it here, causing a double insert.
252 // Because the audio thread and the main thread runs concurrently.
253 bool isFirst = true;
254 while (!audioStream.IsDone()) {
255 AudioClip audioClip = audioStream.TryGetAudioClip();
256 if (audioClip == null) {
257 await Awaitable.WaitForSecondsAsync(0.25f);
258 continue;
259 }
260
261 lock (audioClipGarbageCollector) {
262 audioClipGarbageCollector.Add(audioClip, false);
263 }
264
265 // Actually queue the actor output to be played.
266 lock (actorOutputQueue) {
267 actorOutputQueue.Enqueue(new ActorOutput(isFirst, persona, message, emotion, reasonings, audioClip, audioSource));
268 }
269 isFirst = false;
270 }
271 }

273 /// <summary>
274 /// Launch an audio thread via Coroutine to play any queued audio from the actors.
275 /// </summary>
276 /// <returns>An IEnumerator for Coroutine.</returns>
277 private IEnumerator AudioThread() {
278 // We do not play the audio in the main thread because it is an infinite loop and we don't want to hang the main thread.
279 Debug.Log("Starting audio thread...");
280
281 // Run this loop until the MonoBehaviour is destroyed.
282 AudioSource playingAudioSource = null;
283 while (!destroyCancellationToken.IsCancellationRequested) {
284 // If there is already an audio clip playing, wait for it to be done.
285 if (playingAudioSource != null && playingAudioSource.isPlaying) {
286 yield return null;
287 continue;
288 }
289
290 // Mark this audio clip as finished and ready to be deleted.
291 if (playingAudioSource != null && playingAudioSource.clip != null) {
292 lock (audioClipGarbageCollector) {
293 audioClipGarbageCollector[playingAudioSource.clip] = true;
294 }
295 }
296
297 playingAudioSource = null;
298
299 // If no audio clip is currently playing, try to see if there's any audio clip in the queue.
300 bool hasOutput = false;
301 ActorOutput actorOutput = null;
302 lock (actorOutputQueue) {
303 hasOutput = actorOutputQueue.TryDequeue(out actorOutput);
304 }
305
306 // If there is an audio clip, play it.
307 if (hasOutput) {
308 playingAudioSource = actorOutput.audioSource;
309 playingAudioSource.clip = actorOutput.audioClip;
310 if (actorOutput.audioSource != null && actorOutput.audioClip != null) {
311 playingAudioSource.Play();
312 }
313
314 if (actorOutput.isFirst) {
315 Debug.Log($"{actorOutput.persona} is {actorOutput.emotion.ToString()}");
316 chatbox.AddMessage(actorOutput.persona, actorOutput.message);
317 if (showReasoning) {
318 for (int i = 0; i < actorOutput.reasonings.Count; ++i) {
319 chatbox.AddMessage(actorOutput.persona + $"'s Reasoning {i + 1}", actorOutput.reasonings[i]);
320 }
321 }
322 }
323 }
324 // Otherwise, if there's no more audio clips in the queue, and the main thread isn't queuing any more audio clips, reset the isAudioPlaying flag.
325 else if (!isQueueingAudio) {
326 isAudioPlaying = false;
327 }
328
329 // Wait for next frame to save some CPU cycles.
330 // It's unlikely for an audio clip to finish playing in 1 frame so there's no point checking the if the audio source is done playing.
331 // No point running this loop a bajillion times per frame.
332 yield return null;
333 }
334
335 Debug.Log("Shutting down audio thread...");
336 }

391 discussion.GetTriggerRound() <= roundCounter) {
392 discussionQueue.Enqueue(discussion);
393 untriggeredDiscussions.RemoveAt(i);
394 fsm.ChangeState((int)State.Discuss);

396 }

399 // Otherwise, wait for the director to be ready to listen for user input.
400 fsm.ChangeState((int)State.Wait);

406 private void OnUpdateWait() {
407 if (!isAudioPlaying) {
408 // If no audio is playing and the game should finish, and we did not trigger continue chat, go to the finish state.
409 if (!continueChat && roundCounter >= finishRound) {
410 fsm.ChangeState((int)State.Finish);
411 }
412 // If the director is connected to OpenAI's server and no audio is playing, listen for user input.
413 else if (director.IsConnected && director.IsStatus(Director.Status.Waiting)) {
414 fsm.ChangeState((int)State.Listen);

419 // Listen State
420 private void OnEnterListen() {
421 Debug.Log("VoiceChat: OnEnterListen");
422
423 idleTimer = 0.0f;
424 discussionQueue.Clear();
425 speakerQueue.Clear();
426
427 // Let the director know the name of the actors so that it can determine the speaking order when replying to the user.
428 List<string> speakers = actors.Keys.ToList();
429
430 // Let the director know which discussions it can trigger based on topic.
431 List<string> topics = untriggeredDiscussions.
432 Where(d => d.HasAllTriggerModes(Discussion.TriggerMode.Topic)).
433 Select(d => d.GetTriggerTopic()).ToList();
434
435 // Get the direction to listen for user input.
436 director.ListenForNextUserInput(directorConfig, speakers, topics, destroyCancellationToken);
437 }

439 private void OnUpdateListen() {
440 // Update Idle Timer
441 if (director.IsMicMuted) { idleTimer += Time.deltaTime; }
442 if (idleTimerText != null) { idleTimerText.text = $"Idle Time: {idleTimer.ToString("n2")}s"; }
443
444 // Check if we should trigger any idle discussions if the user has not given any input after a while. This ends the round.
445 for (int i = 0; i < untriggeredDiscussions.Count; ++i) {
446 Discussion discussion = untriggeredDiscussions[i];
447 // Check if we should trigger this idle discussion.
448 if (discussion.HasAllTriggerModes(Discussion.TriggerMode.IdleTime) && discussion.GetTriggerIdleTime() <= idleTimer) {
449 // If the director has already stopped listening, abort.
450 if (!director.CancelListen()) {
451 Debug.Log("Aborting idle discussion as director has already started responding.");
452 break;
453 }
454
455 // Else, trigger idle discussion.
456 discussionQueue.Enqueue(discussion);
457 untriggeredDiscussions.RemoveAt(i);
458 fsm.ChangeState((int)State.Discuss);
459 return;
460 }
461 }

465 private async void RunFreeTalk() {
466 BeginQueuingAudio();
467
468 // Get a response from every actor.
469 while (0 < speakerQueue.Count) {
470 var (actor, audioSource) = actors.GetValueOrDefault(speakerQueue.Dequeue(), (null, null));

580 /// Callback invoked by the director when it has a response ready.
581 /// <param name="response">The director's response.</param>
582 private void OnDirectorResponse(Director.Response response) {
583 // We only care about a response when we are in the listening state.
584 if (fsm.GetCurrentState() != (int)State.Listen) { return; }
585
586 // Add the user transcript to the chatbox.
587 chatbox.AddMessage("User", response.userTranscript);
588 // Inform each actor of what the user said.
589 PropogateUserMessage(response.userTranscript);
590
591 // Try to trigger a discussion.
592 if (response.discussionTopic != null) {
593 for (int i = 0; i < untriggeredDiscussions.Count; ++i) {
594 Discussion discussion = untriggeredDiscussions[i];
595 if (!string.IsNullOrEmpty(discussion.GetTriggerTopic()) &&
596 discussion.GetTriggerTopic() == response.discussionTopic) {
597 discussionQueue.Enqueue(discussion);
598 untriggeredDiscussions.RemoveAt(i);
599 fsm.ChangeState((int)State.Discuss);
600 return;
601 }
602 }
603 }
604
605 // Otherwise, get the actors to respond as per usual.
606 if (response.speakerOrder != null) {
607 foreach (string speaker in response.speakerOrder) {
608 speakerQueue.Enqueue(speaker);
609 }
610 fsm.ChangeState((int)State.FreeTalk);
611 return;
612 }
613
614 // As a backup, just respond in order.
615 foreach (var key in actors.Keys) {
616 speakerQueue.Enqueue(key);
617 fsm.ChangeState((int)State.FreeTalk);
618 Debug.Log("Director decision failed. Applying backup.");
619 }

622 // Configuration File Override
623 private void Override() {
624 string filePath = $"{Application.streamingAssetsPath}/Configs/{OverrideFileName}";
625 IConfiguration config = new ConfigurationBuilder().AddIniFile(filePath).Build();
626 IConfigurationSection section = config.GetSection("VoiceChat");
627
628 // We cannot proceed if the section does not exist.
629 if (section == null) {
630 Debug.LogWarning($"Section VoiceChat not found in {filePath}!");
631 return;
632 }
633
634 // Define a helper function.
635 Func<string, string> GetValue = (string key) => { return section[key] == null ? string.Empty : section[key].Trim(); };
636 string value = string.Empty;
637 int parsedInt = 0;
638
639 // Override OpenAI Vector Store ID
640 value = GetValue("finish_round");
641 if (!string.IsNullOrEmpty(value) && int.TryParse(value, out parsedInt)) {
642 finishRound = parsedInt;
643 Debug.Log($"Overrode VoiceChat's Finish Round to {finishRound}");
644 }
645 }
