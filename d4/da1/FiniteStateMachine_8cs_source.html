3namespace FSM {
4 /// Finite state machine class to handle state transitions and updates.
5 public class FiniteStateMachine {
6 public const int INVALID_STATE = -1; // Use negative value to denote an invalid state.
7
8 public readonly int NumStates = 0;
9 private int currentState = INVALID_STATE;
10 private int nextState = INVALID_STATE;
11
12 /// Define a delegate that takes in 0 arguments and returns void.
13 public delegate void FuncPtr();
14 /// The entry callback function of states.
15 private FuncPtr[] stateEntries = null;
16 /// The update callback function of states.
17 private FuncPtr[] stateUpdates = null;
18 /// The late update callback function of states.
19 private FuncPtr[] stateLateUpdates = null;
20 /// The exit callback function of states.
21 private FuncPtr[] stateExits = null;
22
23 public FiniteStateMachine(int numStates) {
24 // Initialise arrays to hold the function pointers to the entry, update, late update and exit callback functions of the states.
25 NumStates = numStates;
26 stateEntries = new FuncPtr[numStates];
27 stateUpdates = new FuncPtr[numStates];
28 stateLateUpdates = new FuncPtr[numStates];
29 stateExits = new FuncPtr[numStates];
30 }
31
32 public int GetCurrentState() { return currentState; }
33 public int GetNextState() { return nextState; }
34
35 public void SetStateEntry(int state, FuncPtr funcPtr = null) { stateEntries[state] = funcPtr; }
36 public void SetStateUpdate(int state, FuncPtr funcPtr = null) { stateUpdates[state] = funcPtr; }
37 public void SetStateLateUpdate(int state, FuncPtr funcPtr = null) { stateLateUpdates[state] = funcPtr; }
38 public void SetStateExit(int state, FuncPtr funcPtr = null) { stateExits[state] = funcPtr; }
39
40 /// Sets the next state to transit to.
41 /// <param name="nextState">The index of the next state.</param>
42 public void ChangeState(int nextState) { this.nextState = nextState; }
43
44 public void Update() {
45 // Check if the next state is different from the current state.
46 if (nextState != currentState) {
47 // Exit current state.
48 if (0 <= currentState) { stateExits[currentState]?.Invoke(); }
49
50 // Enter next state.
51 currentState = nextState;
52 if (0 <= currentState) { stateEntries[currentState]?.Invoke(); }
53 }
54
55 // Update current state every frame.
56 if (0 <= currentState) { stateUpdates[currentState]?.Invoke(); }
57 }
58
59 public void LateUpdate() {
60 // Late update current state every frame.
61 if (0 <= currentState) { stateLateUpdates[currentState]?.Invoke(); }
62 }
63 }
64}
